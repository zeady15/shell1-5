#!/bin/bash
if [[ $# -lt 1 ]]; then
    echo "Hiba: nem adtal meg parametert!" 1>&2
    exit 1
fi

if [[ ! -e "$1" ]]; then
    echo "Hiba: a fajl nem letezik!" 1>&2
    exit 1
fi

sorok=$(wc -l < "$1")
karakterek=$(wc -c < "$1")

if [[ "$sorok" -eq 0 ]]; then
    echo "Atlagos sorhossz: 0"
else
    atlag=$((karakterek / sorok))
    echo "Atlagos sorhossz: $atlag"
fi
