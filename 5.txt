#!/bin/bash

if [[ $# -lt 1 ]]; then
    echo "Hiba: nem adtak meg állománynevet parancssori paraméterként." 1>&2
    exit 1
fi

file_name="$1"

if [[ ! -e "$file_name" ]]; then
    echo "Hiba: a '$file_name' állomány nem létezik." 1>&2
    exit 1
fi

if [[ ! -f "$file_name" || ! -r "$file_name" ]]; then
    echo "Hiba: a '$file_name' nem olvasható szabályos fájl." 1>&2
    exit 1
fi

line_count=$(wc -l < "$file_name")

if [[ "$line_count" -eq 0 ]]; then
    echo "Az '$file_name' állomány üres (0 sor), átlagos sorhossz: 0"
    exit 0
fi

total_chars=0
while IFS= read -r line; do
    total_chars=$((total_chars + ${#line}))
done < "$file_name"

avg_int=$((total_chars / line_count))

if command -v bc >/dev/null 2>&1; then
    avg_exact=$(echo "scale=2; $total_chars / $line_count" | bc)
    echo "Átlagos sorhossz ('$file_name'): $avg_exact karakter (egészre kerekítve/osztva: $avg_int)"
else
    echo "Átlagos sorhossz ('$file_name'): $avg_int karakter"
fi
