#!/bin/bash

if [[ $# -ne 1 ]]; then
    echo "Hiba: pontosan 1 paramétert (állománynevet) kell megadni, de $# darab érkezett." 1>&2
    exit 1
fi

entry="$1"

if [[ ! -e "$entry" ]]; then
    echo "Hiba: a '$entry' állomány nem létezik." 1>&2
    exit 1
fi

if [[ -d "$entry" ]]; then
    ls -la "$entry"
elif [[ -f "$entry" ]]; then
    echo "!FILE!"
else
    echo "Hiba: a '$entry' nem jegyzék és nem is szabályos fájl." 1>&2
    exit 1
fi
