#!/bin/bash

if [[ $# -ne 2 ]]; then
    echo "Hiba: pontosan 2 paraméter szükséges (fájlnév és minta), de $# darab érkezett." 1>&2
    exit 1
fi

file_name="$1"
pattern="$2"

if [[ ! -e "$file_name" ]]; then
    echo "Hiba: a '$file_name' fájl nem létezik." 1>&2
    exit 1
fi

if [[ ! -r "$file_name" || -d "$file_name" ]]; then
    echo "Hiba: a '$file_name' nem olvasható szövegfájl (vagy mappa)." 1>&2
    exit 1
fi

count=$(grep -E -c "$pattern" "$file_name")

if [[ "$count" -gt 0 ]]; then
    echo "A '$pattern' minta megtalálható a '$file_name' fájlban. Előfordulások száma: $count"
else
    echo "A '$pattern' minta NEM található meg a '$file_name' fájlban."
fi
